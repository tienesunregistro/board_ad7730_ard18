#include <EEPROMex.h>
#include <EEPROMVar.h>
//*************************************************************
//  VALORES POR DEFECTO - EEPROM
//*************************************************************

void FormatearEEPROM()
{
    // write a 0 to all 512 bytes of the EEPROM
    for (int i = 0; i < 512; i++)
        EEPROM.write(i, 0);
}

void SetValorPordefecto(void)
{
    vgEEprom.clave = CLAVE_VERSION;
    vgEEprom.idEstacion = ID_ESTACION;
    vgEEprom.IPE = 1;
    vgEEprom.count_mode = 1;
    vgEEprom.PasoHusillo = 1;
    vgEEprom.PasosEncoder = 1;
    for (int i = 0; i < ID_MAXIMO_CELULA; i++)
    {
        vgEEprom.celulas[i].idcel = i;
        vgEEprom.celulas[i].cap = 1.0;
        vgEEprom.celulas[i].pol = 1;
        vgEEprom.celulas[i].res = 1;
        vgEEprom.celulas[i].limite_carga_celP = 1.0;
        vgEEprom.celulas[i].limite_carga_celN = 1.0;
        vgEEprom.celulas[i].gainpasostoFPos = 1.0;
        vgEEprom.celulas[i].gainpasostoFNeg = 1.0;
        vgEEprom.celulas[i].datarate = DATA_RATE_300; //150;
    }

    vgEEprom.filtro_on_off = 1;    
    vgEEprom.gainpos = 4.0;
    vgEEprom.gainneg = 4.0;
    memset(&vgEEprom.id_maquina, 0x0, sizeof(vgEEprom.id_maquina));
}

void recuperar_valores(void)
{
    EEPROM.readBlock(0, vgEEprom);
    if (vgEEprom.clave != CLAVE_VERSION)
    {

        SetValorPordefecto();
    }
}

void salvar_valores(void)
{
    EEPROM.updateBlock(0, vgEEprom);
    while (EEPROM.isReady() == false)
    {
        delay(1);
    }
    //Serial.println(__FUNCTION__);
    //Serial.println(1);
}
